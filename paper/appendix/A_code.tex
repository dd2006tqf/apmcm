def build_folds(col, row, flood_share, seed=C.FOLD_SEED):
    """Greedy stratified assignment of 5x5-tile blocks to 5 folds."""
    B, K = C.BLOCK, C.N_FOLDS
    bid = (col // B) * 10000 + (row // B)
    ub, inv = np.unique(bid, return_inverse=True)
    barea = np.bincount(inv, weights=flood_share)
    bn = np.bincount(inv)
    rng = np.random.default_rng(seed)
    perm = rng.permutation(len(ub))
    order = perm[np.argsort(-barea[perm], kind="stable")]
    fa, fn = np.zeros(K), np.zeros(K)
    fob = np.zeros(len(ub), int)
    for b in order:
        k = min(range(K), key=lambda k: (fn[k] / 200 + fa[k] / flood_share.sum() * 5))
        fob[b] = k
        fa[k] += barea[b]
        fn[k] += bn[b]
    return fob[inv], inv, len(ub)

class GatedFusionUNet(nn.Module):
    """SAR branch + optional optical branch; at each scale s the fused feature is
    h = g*h_S + (1-g)*h_O with g = sigmoid(W_g * [h_S; h_O]) (1x1 conv)."""
    def __init__(self, c_sar=3, c_opt=7, widths=(16, 32, 64, 128), bottleneck=256, p_drop=0.2):
        super().__init__()
        self.use_opt = c_opt > 0
        self.enc_s = Encoder(c_sar, widths)
        if self.use_opt:
            self.enc_o = Encoder(c_opt, widths)
            self.gates = nn.ModuleList([nn.Conv2d(2 * w, w, 1) for w in widths])
        self.bott = block(widths[-1], bottleneck)
        self.drop = nn.Dropout2d(p_drop)
        self.dec = nn.ModuleList()
        c = bottleneck
        for w in reversed(widths):
            self.dec.append(block(c + w, w))
            c = w
        self.head = nn.Conv2d(widths[0], 1, 1)

    def forward(self, x_sar, x_opt=None, return_gates=False):
        hs = self.enc_s(x_sar)
        if self.use_opt:
            ho = self.enc_o(x_opt)
            skips, gvals = [], []
            for s, gate in enumerate(self.gates):
                g = torch.sigmoid(gate(torch.cat([hs[s], ho[s]], 1)))
                skips.append(g * hs[s] + (1 - g) * ho[s])
                gvals.append(g.mean())
        else:
            skips, gvals = hs, []
        x = self.drop(self.bott(F.max_pool2d(skips[-1], 2)))
        for d, skip in zip(self.dec, reversed(skips)):
            x = F.interpolate(x, size=skip.shape[-2:], mode="bilinear", align_corners=False)
            x = self.drop(d(torch.cat([x, skip], 1)))
        out = self.head(x)[:, 0]
        return (out, gvals) if return_gates else out

def flood_loss(logit, y, valid, lam_area=1.0, gamma=2.0, alpha1=0.75):
    """Focal + soft Dice + lambda_A * area-consistency, valid pixels only."""
    v = valid.float(); p = torch.sigmoid(logit); yf = y.float()
    pt = torch.where(yf > 0.5, p, 1 - p).clamp(1e-6, 1 - 1e-6)
    at = torch.where(yf > 0.5, torch.full_like(p, alpha1), torch.full_like(p, 1 - alpha1))
    focal = (-(at * (1 - pt) ** gamma * torch.log(pt)) * v).sum() / v.sum().clamp(min=1)
    inter = (p * yf * v).sum()
    dice = 1 - (2 * inter + 1) / ((p * v).sum() + (yf * v).sum() + 1)
    nb = v.sum((1, 2)).clamp(min=1)
    area = (((p - yf) * v).sum((1, 2)).abs() / nb).mean()
    return focal + dice + lam_area * area

def fit_isotonic(pbar, y, valid, seed, n=2_000_000):
    vi = torch.nonzero(valid.reshape(-1) > 0).squeeze(1)
    g = torch.Generator(device=vi.device).manual_seed(seed)
    sel = vi[torch.randperm(len(vi), generator=g, device=vi.device)[:n]]
    x = pbar.reshape(-1)[sel].cpu().numpy().astype(np.float64)
    t = y.reshape(-1)[sel].cpu().numpy().astype(np.float64)
    iso = IsotonicRegression(y_min=0, y_max=1, out_of_bounds="clip", increasing=True).fit(x, t)
    return iso

def area_preserving_tau(pbar, wA, expected, lo=0.01, hi=0.99, tol=1e-4):
    """Bisection: sum a*1[p>=tau] = sum a*q(p); left side non-increasing in tau."""
    def f(t):
        return float((wA * (pbar >= t)).sum()) - expected
    if f(lo) < 0 or f(hi) > 0:
        lo, hi = 0.001, 0.999
    while hi - lo > tol:
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    t = lo if abs(f(lo)) <= abs(f(hi)) else hi
    return t, f(t) + expected

def mc_grades(Ahat, sigp, Areg, resid_pool, boot_theta, M=C.MC_GRADE, seed=C.SEED):
    """Grade probabilities pi_jg for units j (one shared residual per draw)."""
    rng = np.random.default_rng(seed)
    r = rng.choice(resid_pool, M, replace=True)
    th = boot_theta[rng.integers(0, len(boot_theta), M)]
    A = np.clip(Ahat[:, None] + r[None, :] * sigp[:, None], 0, Areg[:, None])
    rho = A / Areg[:, None]
    g = 1 + (rho[:, :, None] >= th[None, :, :]).sum(2)
    pi = np.stack([(g == gg).mean(1) for gg in (1, 2, 3, 4)], 1)
    return pi

def coreg_shift(raw, ref, k=4, n_pts=120_000):
    """Integer (dr, dc) maximizing SAR darkness on the permanent-water reference:
    FFT cross-correlation on a 4x decimated grid, then a +-6 px refinement."""
    rng = np.random.default_rng(C.SEED)
    rr, cc = np.nonzero(ref)
    sel = rng.choice(len(rr), min(n_pts, len(rr)), replace=False)
    rr, cc = rr[sel], cc[sel]
    a = ref[::k, ::k].astype(float)
    b = (np.nan_to_num(np.asarray(raw[::k, ::k]), nan=0) < -15).astype(float)
    a, b = a - a.mean(), b - b.mean()
    c = np.fft.irfft2(np.fft.rfft2(a) * np.conj(np.fft.rfft2(b)), s=a.shape)
    i, j = np.unravel_index(np.argmax(c), c.shape)
    i = i if i < a.shape[0] // 2 else i - a.shape[0]
    j = j if j < a.shape[1] // 2 else j - a.shape[1]
    cands = [(sr * i * k, sc * j * k) for sr in (1, -1) for sc in (1, -1)]
    dr0, dc0 = max(cands, key=lambda t: water_hit(raw, rr, cc, *t))
    best = max(((dr0 + x, dc0 + y) for x in range(-6, 7) for y in range(-6, 7)),
               key=lambda t: water_hit(raw, rr, cc, *t))
    return dict(dr=int(best[0]), dc=int(best[1]), score_before=water_hit(raw, rr, cc, 0, 0),
                score_after=water_hit(raw, rr, cc, *best))

def compute_F(meta, date, tauA, dtau=0.0, hstar=C.REM_MAX, delta=C.DELTA_DB,
              w0mode=C.W0_MODE, mmu=C.MMU_PX_S1, scale="db", want_stages=False):
    """F_d = MMU( {p>=tau_A+dtau} & ~W0 & {REM<=h*} & {dVV<=-delta} ) inside the boundary."""
    shape = tuple(meta["shape"]); Hh, Ww = shape
    PB = mmap(f"pbar_{date}_{scale}.f2", np.float16, shape)
    W0 = mmap("W0_union.u1" if w0mode == "union" else "W0_sar.u1", np.uint8, shape)
    bm = mmap("boundary.u1", np.uint8, shape)
    lf_d, lf_0 = mmap(f"vvlee_{date}.f4", np.float32, shape), mmap(f"vvlee_{C.PRE_DATE}.f4", np.float32, shape)
    A = area_raster(meta)
    rem30 = np.load(CQ3 / "rem30.npy")
    cand = np.zeros(shape, bool); stages = np.zeros(4)
    for r in range(0, Hh, 512):
        sl = slice(r, min(r + 512, Hh))
        inb = bm[sl] > 0
        raw = (PB[sl] >= tauA + dtau) & inb
        a = A[sl]
        c1 = raw & (W0[sl] == 0)
        rem = rem10_strip(rem30, r, sl.stop - r, Ww)
        c2 = c1 & (np.nan_to_num(rem, nan=1e9) <= hstar)
        dvv = np.nan_to_num(lf_d[sl] - lf_0[sl], nan=0.0)
        c3 = c2 & (dvv <= -delta)
        cand[sl] = c3
        if want_stages:
            stages += [float(a[raw].sum()), float(a[c1].sum()), float(a[c2].sum()), float(a[c3].sum())]
    F = U.mmu_filter(cand, mmu) if mmu > 1 else cand
    return (F, stages) if want_stages else F

def water_states(f, w0, d0, dd):
    """W1 and receded water R with observed recession only."""
    R = w0 & d0 & ~dd & ~f
    W1 = f | (w0 & ~R)
    return W1, R

\begin{lstlisting}[language=Python]
def entropy_w(R):
    n = R.shape[0]
    P = R / np.where(R.sum(0) > 0, R.sum(0), 1)
    with np.errstate(divide="ignore", invalid="ignore"):
        e = -np.nansum(np.where(P > 0, P * np.log(P), 0), 0) / np.log(n)
    d = 1 - e
    return d / d.sum()

def critic_w(R):
    s = R.std(0, ddof=1)
    cr = np.nan_to_num(np.atleast_2d(np.corrcoef(R, rowvar=False)), nan=0.0)
    c = s * (1 - cr).sum(1)
    return c / c.sum() if c.sum() > 0 else np.full(R.shape[1], 1 / R.shape[1])

def game_combine(we, wc):
    A = np.array([[we @ we, we @ wc], [wc @ we, wc @ wc]])
    b = np.array([we @ we, wc @ wc])
    try:
        al = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        al = np.array([0.5, 0.5])
    w = (abs(al[0]) * we + abs(al[1]) * wc) / (abs(al[0]) + abs(al[1]))
    return w / w.sum(), al

def ahp():
    J = np.array([[1, 2, 3], [1 / 2, 1, 2], [1 / 3, 1 / 2, 1]])
    vals, vecs = np.linalg.eig(J)
    k = np.argmax(vals.real)
    w = np.abs(vecs[:, k].real); w /= w.sum()
    lam = vals[k].real
    return w, lam, (lam - 3) / 2 / 0.58

def topsis(R, w):
    V = R * w
    vp, vm = V.max(0), V.min(0)
    dp = np.sqrt(((V - vp) ** 2).sum(1)); dm = np.sqrt(((V - vm) ** 2).sum(1))
    return np.where(dp + dm > 0, dm / (dp + dm), 0.0)

\subsection{Problem 5: lexicographic location--allocation MILP}

\begin{lstlisting}[language=Python]
def solve(inp, stage="lex", eps=None, fix_eta=None, time_limit=300):
    p, dem, cand, tt, cap = inp["p"], inp["dem"], inp["cand"], inp["tt"], inp["cap"]
    nI, nJ = tt.shape
    d, w = dem["d"].values, dem["w"].values
    Wd = float((w * d).sum())
    A = [(i, j) for i in range(nI) for j in range(nJ) if tt[i, j] <= p["t_max"]]

    def build(obj):
        m = pulp.LpProblem("relief", pulp.LpMinimize)
        eta = {i: pulp.LpVariable(f"eta_{i}", cat=pulp.LpBinary) for i in range(nI)}
        f = {(i, j): pulp.LpVariable(f"f_{i}_{j}", lowBound=0) for (i, j) in A}
        u = {j: pulp.LpVariable(f"u_{j}", lowBound=0) for j in range(nJ)}
        f1 = pulp.lpSum(w[j] * u[j] for j in range(nJ)) * (1.0 / Wd)
        f2 = pulp.lpSum(w[j] * tt[i, j] * f[i, j] for (i, j) in A) * (1.0 / Wd)
        byj, byi = {}, {}
        for (i, j) in A:
            byj.setdefault(j, []).append(f[i, j]); byi.setdefault(i, []).append(f[i, j])
        for j in range(nJ):                                              # demand balance
            m += pulp.lpSum(byj.get(j, [])) + u[j] == d[j], f"bal_{j}"
        for i in range(nI):                                              # capacity, open depots only
            m += pulp.lpSum(byi.get(i, [])) <= cap[i] * eta[i], f"cap_{i}"
            if not cand["admissible"].iloc[i]:
                m += eta[i] == 0, f"excl_{i}"
        for (i, j) in A:                                                 # valid tightening
            m += f[i, j] <= d[j] * eta[i], f"tight_{i}_{j}"
        m += pulp.lpSum(eta.values()) <= p["K"], "K"
        m += pulp.lpSum((2 * tt[i, j] + p["t_load"]) / p["q_veh"] * f[i, j] for (i, j) in A) \
            <= 60 * p["n_veh"] * p["shift"] * p["H"], "fleet"
        return m, eta, f, u, f1, f2

    def run(m):
        m.solve(pulp.HiGHS(msg=False, timeLimit=time_limit, gapRel=0, threads=1))
        return pulp.LpStatus[m.status]

    # lexicographic: stage 1 min f1, stage 2 min f2 s.t. f1 <= f1* + 1e-6
    m, eta, f, u, f1, f2 = build(None)
    m.setObjective(f1); run(m); f1s = pulp.value(f1)
    m, eta, f, u, f1, f2 = build(None)
    m += f1 <= f1s + 1e-6, "lex"
    m.setObjective(f2); run(m)
    return dict(f1=pulp.value(f1), f2=pulp.value(f2),
                eta=[int(round(eta[i].value() or 0)) for i in range(nI)],
                gap=0.0)
